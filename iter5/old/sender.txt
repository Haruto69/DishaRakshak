/**
 * wsn_sender.ino
 * Secure ESP32 + MPU6050 sender node (foot wearable).
 *
 * Fixes applied:
 *  - Force fixed Wi‑Fi channel before esp_now_init()
 *  - Set peer.channel to the fixed channel (no channel=0)
 *  - Avoid Serial in receive callback (ISR context)
 *  - Print WiFi.channel() for runtime verification
 *
 * Notes:
 *  - BASE_MAC, LMK_NODE1/2, WSN_WIFI_CHANNEL, ESP_NOW_PMK, HMAC_KEY,
 *    PKT_TYPE_*, imu_packet_t, ack_packet_t, sync_packet_t, TDMA_* etc.
 *    are expected to be defined in wsn_config.h (single source of truth).
 *  - Do NOT duplicate BASE_MAC in this file; use the one from wsn_config.h.
 */

#include <Arduino.h>
#include <WiFi.h>
#include <esp_now.h>
#include <esp_wifi.h>
#include <Wire.h>
#include <esp_task_wdt.h>
#include "mbedtls/md.h"
#include "wsn_config.h"

// ── Node identity ───────────────
#ifndef NODE_ID
  #define NODE_ID 1
#endif

// ── I2C pins ─────────────────────
#define I2C_SDA  21
#define I2C_SCL  22
#define MPU_ADDR 0x68

#if NODE_ID == 1
  #define TDMA_SLOT_MS  TDMA_SLOT_NODE1_MS
#elif NODE_ID == 2
  #define TDMA_SLOT_MS  TDMA_SLOT_NODE2_MS
#else
  #error "NODE_ID must be 1 or 2"
#endif

// ── State ────────────────────────
static imu_packet_t s_tx_pkt;
static uint32_t     s_seq = 0;

static volatile int64_t  s_time_offset_us = 0;
static volatile bool     s_synced = false;

static volatile bool     s_ack_received = false;
static volatile uint32_t s_ack_seq      = 0;

static volatile bool     s_unknown_mac_flag = false;

static esp_now_peer_info_t      s_peer;
static mbedtls_md_context_t     s_hmac_ctx;
static const mbedtls_md_info_t* s_md_info = nullptr;

// ── Utility: compare MACs ───────────────────────────────
static bool mac_equal(const uint8_t* a, const uint8_t* b) {
    for (int i = 0; i < 6; i++) if (a[i] != b[i]) return false;
    return true;
}

// ═════════════════════════════════════════════════════════════
// SECURITY – HMAC-SHA256
// ═════════════════════════════════════════════════════════════
static void security_init() {
    mbedtls_md_init(&s_hmac_ctx);
    s_md_info = mbedtls_md_info_from_type(MBEDTLS_MD_SHA256);
    if (!s_md_info) {
        Serial.println("[SEC] FATAL: SHA256 info null");
        while (true) {}
    }
    if (mbedtls_md_setup(&s_hmac_ctx, s_md_info, /*hmac=*/1) != 0) {
        Serial.println("[SEC] FATAL: md_setup failed");
        while (true) {}
    }
}
static void security_compute_hmac(imu_packet_t* pkt) {
    const size_t auth_len = sizeof(imu_packet_t) - sizeof(pkt->hmac);
    mbedtls_md_hmac_starts(&s_hmac_ctx, HMAC_KEY, sizeof(HMAC_KEY));
    mbedtls_md_hmac_update(&s_hmac_ctx,
                           reinterpret_cast<const uint8_t*>(pkt),
                           auth_len);
    mbedtls_md_hmac_finish(&s_hmac_ctx, pkt->hmac);
}

// ═════════════════════════════════════════════════════════════
// SENSOR – MPU6050 (bare-metal I2C)
static void mpu_write(uint8_t reg, uint8_t val) {
    Wire.beginTransmission(MPU_ADDR);
    Wire.write(reg);
    Wire.write(val);
    Wire.endTransmission();
}
static void sensor_init() {
    Wire.begin(I2C_SDA, I2C_SCL);
    delay(100);

    mpu_write(0x6B, 0x00);  // PWR_MGMT_1: wake
    mpu_write(0x19, 0x09);  // SMPLRT_DIV: 100 Hz
    mpu_write(0x1A, 0x03);  // CONFIG: DLPF_CFG=3
    mpu_write(0x1B, 0x00);  // GYRO_CONFIG: ±250 °/s
    mpu_write(0x1C, 0x00);  // ACCEL_CONFIG: ±2 g

    // Verify WHO_AM_I.
    Wire.beginTransmission(MPU_ADDR);
    Wire.write(0x75);
    Wire.endTransmission(false);
    Wire.requestFrom(MPU_ADDR, (uint8_t)1);
    if (Wire.available()) {
        uint8_t who = Wire.read();
        if (who != 0x68) {
            Serial.printf("[SENSOR] WARN: WHO_AM_I=0x%02X (expected 0x68)\n", who);
        } else {
            Serial.println("[SENSOR] MPU6050 OK");
        }
    } else {
        Serial.println("[SENSOR] WARN: WHO_AM_I read failed");
    }
}
static void sensor_read(imu_packet_t* pkt) {
    Wire.beginTransmission(MPU_ADDR);
    Wire.write(0x3B);  // ACCEL_XOUT_H
    Wire.endTransmission(false);
    Wire.requestFrom(MPU_ADDR, (uint8_t)14);

    pkt->accel[0] = (int16_t)((Wire.read() << 8) | Wire.read());
    pkt->accel[1] = (int16_t)((Wire.read() << 8) | Wire.read());
    pkt->accel[2] = (int16_t)((Wire.read() << 8) | Wire.read());
    Wire.read(); Wire.read(); // TEMP skip
    pkt->gyro[0]  = (int16_t)((Wire.read() << 8) | Wire.read());
    pkt->gyro[1]  = (int16_t)((Wire.read() << 8) | Wire.read());
    pkt->gyro[2]  = (int16_t)((Wire.read() << 8) | Wire.read());
}

// ═════════════════════════════════════════════════════════════
// COMMUNICATIONS – ESP-NOW
// Receive callback runs in Wi‑Fi task context; avoid Serial here.
static void comms_on_recv(const esp_now_recv_info_t* info,
                          const uint8_t* data, int len)
{
    // Only accept packets from the base station (BASE_MAC defined in wsn_config.h)
    if (!mac_equal(info->src_addr, BASE_MAC)) {
        s_unknown_mac_flag = true; // set flag; print in loop()
        return;
    }

    if (len < 1) return;
    uint8_t pkt_type = data[0];

    // ACK packet
    if (pkt_type == PKT_TYPE_ACK && len == sizeof(ack_packet_t)) {
        const ack_packet_t* ack = reinterpret_cast<const ack_packet_t*>(data);
        if (ack->node_id == NODE_ID) {
            s_ack_seq      = ack->ack_seq;
            s_ack_received = true;
        }
        return;
    }

    // SYNC packet
    if (pkt_type == PKT_TYPE_SYNC && len == sizeof(sync_packet_t)) {
        const sync_packet_t* sp = reinterpret_cast<const sync_packet_t*>(data);
        int64_t local_now  = (int64_t)esp_timer_get_time();
        s_time_offset_us   = (int64_t)sp->base_time_us - local_now;
        s_synced           = true;
        return;
    }
}

// comms_init: enforce channel lock BEFORE esp_now_init(), set PMK, register peer
static void comms_init() {
    WiFi.mode(WIFI_STA);
    WiFi.disconnect();

    // Force fixed Wi-Fi channel before esp_now_init()
    esp_wifi_set_promiscuous(true);
    esp_wifi_set_channel(WSN_WIFI_CHANNEL, WIFI_SECOND_CHAN_NONE);
    esp_wifi_set_promiscuous(false);

    if (esp_now_init() != ESP_OK) {
        Serial.println("[COMMS] FATAL: esp_now_init failed");
        while (true) {}
    }

    // Set PMK and register receive callback
    esp_now_set_pmk(ESP_NOW_PMK);
    esp_now_register_recv_cb(comms_on_recv);

    // Prepare peer entry for base station (BASE_MAC comes from wsn_config.h)
    memset(&s_peer, 0, sizeof(s_peer));
    memcpy(s_peer.peer_addr, BASE_MAC, 6);
    s_peer.channel = WSN_WIFI_CHANNEL;   // MUST match forced channel (not 0)
    s_peer.encrypt = true;

#if NODE_ID == 1
    memcpy(s_peer.lmk, LMK_NODE1, 16);
#else
    memcpy(s_peer.lmk, LMK_NODE2, 16);
#endif

    if (esp_now_add_peer(&s_peer) != ESP_OK) {
        Serial.println("[COMMS] FATAL: add_peer failed");
        while (true) {}
    }

    Serial.print("Current Channel: ");
    Serial.println(WiFi.channel());   // runtime verification
    Serial.println("[COMMS] ESP-NOW initialised");
}

// Send with retry (keeps existing retry logic)
static void comms_send_with_retry(imu_packet_t* pkt) {
    for (int attempt = 0; attempt <= MAX_RETRANSMIT; attempt++) {
        s_ack_received = false;

        esp_err_t err = esp_now_send(BASE_MAC,
                                     reinterpret_cast<uint8_t*>(pkt),
                                     sizeof(imu_packet_t));
        if (err != ESP_OK) {
            Serial.printf("[COMMS] send err=0x%X attempt=%d\n", err, attempt);
            continue;
        }

        // Busy-wait for ACK with watchdog resets
        uint32_t t0 = millis();
        while (!s_ack_received && (millis() - t0) < ACK_TIMEOUT_MS) {
            delayMicroseconds(200);
            esp_task_wdt_reset();
        }

        if (s_ack_received && s_ack_seq == pkt->seq) return; // Success

        Serial.printf("[COMMS] No ACK for seq=%u (attempt %d)\n",
                      pkt->seq, attempt);
    }
    Serial.printf("[COMMS] Dropped seq=%u\n", pkt->seq);
}

// ═════════════════════════════════════════════════════════════
// TDMA scheduler (unchanged)
static void tdma_wait_for_slot() {
    int64_t now_us     = (int64_t)esp_timer_get_time() + s_time_offset_us;
    int64_t now_ms     = now_us / 1000LL;
    int64_t cycle_ms   = (int64_t)TDMA_CYCLE_MS;
    int64_t slot_ms    = (int64_t)TDMA_SLOT_MS;
    int64_t next_cycle = ((now_ms / cycle_ms) + 1) * cycle_ms;
    int64_t target_ms  = next_cycle + slot_ms;
    int64_t wait_us    = (target_ms * 1000LL) - now_us;

    if (wait_us > 0 && wait_us < (int64_t)TDMA_CYCLE_MS * 2000LL) {
        delayMicroseconds((uint32_t)wait_us);
    }
    esp_task_wdt_reset();
}

// ═════════════════════════════════════════════════════════════
// ARDUINO ENTRY POINTS
void setup() {
    Serial.begin(115200);
    delay(200);
    Serial.printf("\n[WSN] Sender Node %d starting...\n", NODE_ID);

    // Watchdog
    esp_task_wdt_add(NULL);

    // Init modules
    security_init();
    sensor_init();
    comms_init();   // channel locked inside here

    // Print own MAC so you can verify / update wsn_config.h
    Serial.print("[WSN] Sender MAC: ");
    WiFi.mode(WIFI_STA);
    Serial.println(WiFi.macAddress());

    // Pre-fill static packet fields
    memset(&s_tx_pkt, 0, sizeof(s_tx_pkt));
    s_tx_pkt.pkt_type = PKT_TYPE_IMU;
    s_tx_pkt.node_id  = NODE_ID;

    // Wait for first SYNC beacon before transmitting (FIX-S2)
    Serial.println("[WSN] Waiting for first SYNC beacon from base station...");
    while (!s_synced) {
        esp_task_wdt_reset();
        delayMicroseconds(500);
    }
    Serial.println("[WSN] SYNC received – starting transmission");

    Serial.printf("[WSN] Node %d ready. TDMA slot = %d ms\n",
                  NODE_ID, TDMA_SLOT_MS);
}

void loop() {
    esp_task_wdt_reset();

    // Print rejected-MAC warning safely from loop context
    if (s_unknown_mac_flag) {
        s_unknown_mac_flag = false;
        Serial.println("[COMMS] Rejected: packet from unknown source MAC");
    }

    // Wait for our TDMA transmission slot
    tdma_wait_for_slot();

    // Read IMU
    sensor_read(&s_tx_pkt);

    // Fill metadata
    s_tx_pkt.seq = ++s_seq;
    s_tx_pkt.timestamp_us =
        (uint64_t)((int64_t)esp_timer_get_time() + s_time_offset_us);

    // Compute HMAC last
    security_compute_hmac(&s_tx_pkt);

    // Transmit with ACK/retry
    comms_send_with_retry(&s_tx_pkt);

    // Debug heartbeat every 50 packets (~0.5 s at 100 Hz)
    if (s_seq % 50 == 0) {
        Serial.printf("[WSN] seq=%u ts=%llu ax=%d ay=%d az=%d\n",
                      s_tx_pkt.seq,
                      (unsigned long long)s_tx_pkt.timestamp_us,
                      s_tx_pkt.accel[0],
                      s_tx_pkt.accel[1],
                      s_tx_pkt.accel[2]);
    }
}
