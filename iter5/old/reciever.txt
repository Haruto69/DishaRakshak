#include <Arduino.h>
#include <WiFi.h>
#include <esp_now.h>
#include <esp_wifi.h>
#include <esp_task_wdt.h>
#include "wsn_config.h"

// ── Base station MAC (this device’s MAC, update in wsn_config.h) ─────────
// Do NOT redeclare BASE_MAC here if it’s already defined in wsn_config.h

// ── State ───────────────────────────────────────────────────
static volatile bool s_ack_pending = false;
static volatile uint32_t s_last_seq = 0;

// ═════════════════════════════════════════════════════════════
// COMMUNICATIONS – ESP-NOW
// ═════════════════════════════════════════════════════════════
static void comms_on_recv(const esp_now_recv_info_t* info,
                          const uint8_t* data, int len)
{
    if (len < 1) return;
    uint8_t pkt_type = data[0];

    // IMU packet received
    if (pkt_type == PKT_TYPE_IMU && len == sizeof(imu_packet_t)) {
        const imu_packet_t* pkt = reinterpret_cast<const imu_packet_t*>(data);
        s_last_seq = pkt->seq;
        s_ack_pending = true;

        Serial.printf("[RX] IMU seq=%u ax=%d ay=%d az=%d\n",
                      pkt->seq, pkt->accel[0], pkt->accel[1], pkt->accel[2]);
    }
}

static void comms_init() {
    WiFi.mode(WIFI_STA);
    WiFi.disconnect();

    // 🔴 Force fixed Wi-Fi channel before ESP-NOW init
    esp_wifi_set_promiscuous(true);
    esp_wifi_set_channel(WSN_WIFI_CHANNEL, WIFI_SECOND_CHAN_NONE);
    esp_wifi_set_promiscuous(false);

    if (esp_now_init() != ESP_OK) {
        Serial.println("[COMMS] FATAL: esp_now_init failed");
        while (true) {}
    }

    esp_now_set_pmk(ESP_NOW_PMK);
    esp_now_register_recv_cb(comms_on_recv);

    // Register sender peers (update MACs with actual sender addresses)
    esp_now_peer_info_t peerInfo;
    memset(&peerInfo, 0, sizeof(peerInfo));

    uint8_t NODE1_MAC[] = {0x70, 0x4B, 0xCA, 0x47, 0x6D, 0x3C}; // example sender MAC
    memcpy(peerInfo.peer_addr, NODE1_MAC, 6);
    peerInfo.channel = WSN_WIFI_CHANNEL;
    peerInfo.encrypt = true;
    memcpy(peerInfo.lmk, LMK_NODE1, 16);
    esp_now_add_peer(&peerInfo);

    Serial.print("Current Channel: ");
    Serial.println(WiFi.channel());
    Serial.println("[COMMS] Receiver initialised");
}

// ═════════════════════════════════════════════════════════════
// ACK HANDLING
// ═════════════════════════════════════════════════════════════
static void comms_send_ack(uint32_t seq, uint8_t node_id) {
    ack_packet_t ack;
    ack.pkt_type = PKT_TYPE_ACK;
    ack.node_id  = node_id;
    ack.ack_seq  = seq;

    esp_err_t err = esp_now_send(BASE_MAC,
                                 reinterpret_cast<uint8_t*>(&ack),
                                 sizeof(ack));
    if (err != ESP_OK) {
        Serial.printf("[ACK] send err=0x%X\n", err);
    } else {
        Serial.printf("[ACK] Sent for seq=%u\n", seq);
    }
}

// ═════════════════════════════════════════════════════════════
// SYNC HANDLING – send periodic SYNC packets
// ═════════════════════════════════════════════════════════════
static void comms_send_sync() {
    sync_packet_t sp;
    sp.pkt_type     = PKT_TYPE_SYNC;
    sp.base_time_us = (uint64_t)esp_timer_get_time();

    esp_err_t err = esp_now_send(BASE_MAC, (uint8_t*)&sp, sizeof(sp));
    if (err == ESP_OK) {
        Serial.println("[SYNC] SYNC packet sent");
    } else {
        Serial.printf("[SYNC] send error=0x%X\n", err);
    }
}

// ═════════════════════════════════════════════════════════════
// ARDUINO ENTRY POINTS
// ═════════════════════════════════════════════════════════════
void setup() {
    Serial.begin(115200);
    delay(200);
    Serial.println("[WSN] Receiver starting...");

    esp_task_wdt_add(NULL);
    comms_init();
}

void loop() {
    esp_task_wdt_reset();

    // Send ACK if needed
    if (s_ack_pending) {
        s_ack_pending = false;
        comms_send_ack(s_last_seq, NODE_ID);
    }

    // Send SYNC every 1 second
    static uint32_t lastSyncMs = 0;
    if (millis() - lastSyncMs > 1000) {
        lastSyncMs = millis();
        comms_send_sync();
    }

    delay(10);
}
