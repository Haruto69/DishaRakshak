 *        #include <WiFi.h>
 *        void setup() { Serial.begin(115200); WiFi.mode(WIFI_STA); Serial.println(WiFi.macAddress()); }
 *        void loop() {}
 *   2. Fill in BASE_MAC, NODE1_MAC, NODE2_MAC with the real values.
 *   3. Change all key material (PMK, LMK_*, HMAC_KEY) for production.
 */

// ─────────────────────────────────────────────────────────────
// TIMING CONSTANTS
// ─────────────────────────────────────────────────────────────

#define SAMPLE_RATE_HZ            100          // MPU6050 sample rate
#define SAMPLE_INTERVAL_US        10000UL      // 1 000 000 / SAMPLE_RATE_HZ

#define TDMA_CYCLE_MS             10           // Full TDMA cycle length
#define TDMA_SLOT_NODE1_MS        0            // Node 1 TX offset within cycle
#define TDMA_SLOT_NODE2_MS        5            // Node 2 TX offset within cycle

#define SYNC_INTERVAL_MS          1000         // Base-station sync beacon period
#define ACK_TIMEOUT_MS            20           // Retransmit if no ACK within N ms
#define MAX_RETRANSMIT            1            // Retransmit once, then drop

#define MAX_TIMESTAMP_DRIFT_US    50000UL      // 50 ms tolerance
#define WATCHDOG_TIMEOUT_MS       3000         // WDT reset threshold

// ─────────────────────────────────────────────────────────────
// PACKET TYPES
// ─────────────────────────────────────────────────────────────
#define PKT_TYPE_IMU              0x01
#define PKT_TYPE_ACK              0x02
#define PKT_TYPE_SYNC             0x03

// ─────────────────────────────────────────────────────────────
// PACKET STRUCTURES  (packed – no padding)
// ─────────────────────────────────────────────────────────────
#pragma pack(push, 1)

// IMU data packet (sent by each node).
typedef struct {
    uint8_t  pkt_type;       // PKT_TYPE_IMU
    uint8_t  node_id;        // 1 or 2
    uint32_t seq;            // Strictly increasing sequence number
    uint64_t timestamp_us;   // Synchronized timestamp (µs)
    int16_t  accel[3];       // ax, ay, az  (raw ADC units)
    int16_t  gyro[3];        // gx, gy, gz  (raw ADC units)
    uint8_t  hmac[32];       // HMAC-SHA256 over all fields above
} imu_packet_t;              // Total: 1+1+4+8+6*2+6*2+32 = 58 bytes

// ACK packet (sent by base station).
typedef struct {
    uint8_t  pkt_type;       // PKT_TYPE_ACK
    uint8_t  node_id;        // Which node this ACK is for
    uint32_t ack_seq;        // Sequence number being acknowledged
} ack_packet_t;              // 6 bytes

// Sync packet (sent by base station every SYNC_INTERVAL_MS).
typedef struct {
    uint8_t  pkt_type;       // PKT_TYPE_SYNC
    uint64_t base_time_us;   // Base station absolute time (µs)
} sync_packet_t;             // 9 bytes

#pragma pack(pop)

// Compile-time size assertions.
static_assert(sizeof(imu_packet_t)  == 58, "imu_packet_t size mismatch");
static_assert(sizeof(ack_packet_t)  ==  6, "ack_packet_t size mismatch");
static_assert(sizeof(sync_packet_t) ==  9, "sync_packet_t size mismatch");
